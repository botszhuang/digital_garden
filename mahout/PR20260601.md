- 1.移除 if的 control-flow，除去 branch divergence。一個Wrap的32 個執行緒的 FMA 硬體單元同時啟動，在一兩個時脈週期內一起算完。整齊劃一，沒有任何執行緒需要被凍結。
- 2.使用 CUDA 的 pow(x, y) 函式透過底層的專用數學硬體單元（SFU, Special Function Unit）來計算。這執行速度非常快（通常只需幾個時脈週期），但因為涉及對數與指數的轉換，在極端情況下可能會引入極其微小的浮點數精度誤差（通常在最後幾個 bits）
```c
__global__ void phase_encode_kernel(
    const double* __restrict__ phases,
    cuDoubleComplex* __restrict__ state,
    size_t state_len,
    unsigned int num_qubits
) {
    size_t idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= state_len) return;

    // φ(idx) = Σ_k phases[k] * b_k,  b_k = (idx >> k) & 1
    double phi = 0.0;
    for (unsigned int bit = 0; bit < num_qubits; ++bit) {
        /* original code:
        if ((idx >> bit) & 1U) {
            phi += phases[bit];
        }*/
        // ===== 1 ======
        phi += phases[bit]*((idx >> bit) & 1U)); 
    }

    // original code:
    // double norm = phase_norm(num_qubits);
    // ===== 2 =====
    double norm = pow(M_SQRT1_2, num_qubits);

    double re, im;
    sincos(phi, &im, &re);   // re = cos(φ), im = sin(φ)

    state[idx] = make_cuDoubleComplex(norm * re, norm * im);

}
```
